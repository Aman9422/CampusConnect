## Firestore Schema

### New Collections

#### `mentorship_requests/`
```javascript
{
  studentId: string,              // Requesting student's UID
  alumniId: string,              // Target alumni's UID
  title: string,                 // Request title
  description: string,           // Detailed request description
  skills: string[],              // Requested mentorship skills
  status: string,                // 'pending'|'accepted'|'rejected'|'completed'
  createdAt: timestamp,          // Request creation time
  respondedAt?: timestamp,       // Alumni response time
  responseMessage?: string,      // Alumni response message
  studentName: string,           // Cached student name
  studentEmail: string,          // Cached student email
  alumniName: string,           // Cached alumni name
  alumniCompany?: string        // Cached alumni company
}
```

#### `opportunities/`
```javascript
{
  alumniId: string,              // Posting alumni's UID
  title: string,                 // Job title
  company: string,               // Company name
  description: string,           // Job description
  requirements: string[],        // Job requirements
  location: string,              // Job location
  jobType: string,              // 'full-time'|'part-time'|'internship'|'contract'
  skills: string[],             // Required skills
  postedAt: timestamp,          // Posting time
  applicationDeadline?: timestamp, // Application deadline
  isActive: boolean,            // Active status
  salaryRange?: string,         // Salary information
  applicationUrl?: string,      // External application link
  contactEmail?: string,        // Contact information
  alumniName: string,          // Cached alumni name
  alumniEmail: string          // Cached alumni email
}
```

### Security Rules Added
```javascript
// Mentorship requests - students create, alumni respond
match /mentorship_requests/{requestId} {
  allow read: if isAuthenticated() &&
    (resource.data.studentId == request.auth.uid ||
     resource.data.alumniId == request.auth.uid);
  allow create: if isAuthenticated() &&
    request.resource.data.studentId == request.auth.uid;
  allow update: if isAuthenticated() &&
    resource.data.alumniId == request.auth.uid &&
    request.resource.data.diff(resource.data).affectedKeys()
    .hasOnly(['status', 'respondedAt', 'responseMessage']);
}

// Job opportunities - alumni create, all authenticated users can read
match /opportunities/{opportunityId} {
  allow read: if isAuthenticated();
  allow create: if isAuthenticated() &&
    request.resource.data.alumniId == request.auth.uid;
  allow update, delete: if isAuthenticated() &&
    resource.data.alumniId == request.auth.uid;
}
```
